import { serve } from "https://deno.land/std@0.224.0/http/server.ts"; import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; type Json = Record; const json = (s: number, b: Json) => new Response(JSON.stringify(b), { status: s, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); serve(async (req) => { try { if (req.method === "OPTIONS") { return new Response(null, { status: 204, headers: corsHeaders }); } if (req.method !== "POST") return json(405, { ok: false, error: "Method not allowed" }); const SUPABASE_URL = Deno.env.get("SUPABASE_URL"); const ANON = Deno.env.get("SUPABASE_ANON_KEY"); // Editable key (new Supabase key system) const SR = Deno.env.get("SERVICE_ROLE_KEY") ?? Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); if (!SUPABASE_URL || !ANON || !SR) { return json(500, { ok: false, error: "Server misconfiguration" }); } // Caller identity (JWT from door.html) const supabaseAuth = createClient(SUPABASE_URL, ANON, { global: { headers: { Authorization: req.headers.get("Authorization") ?? "" } }, }); const { data: { user }, error: authErr } = await supabaseAuth.auth.getUser(); if (authErr || !user) return json(401, { ok: false, error: "Unauthorized" }); const emailRaw = user.email; if (!emailRaw) return json(400, { ok: false, error: "User email missing" }); const email = emailRaw.trim().toLowerCase(); // Admin client (RPC + allowlists) const admin = createClient(SUPABASE_URL, SR); // Allowlist OR gate (presence = access) const [orgRes, doorRes] = await Promise.all([ admin.from("organizer_allowlist").select("email").eq("email", email).maybeSingle(), admin.from("door_staff_allowlist").select("email").eq("email", email).maybeSingle(), ]); if (orgRes.error || doorRes.error) { return json(500, { ok: false, error: "Allowlist lookup failed", details: { organizer_error: orgRes.error?.message ?? null, door_error: doorRes.error?.message ?? null, }, }); } if (!orgRes.data && !doorRes.data) { return json(403, { ok: false, error: "Forbidden" }); } const body = await req.json().catch(() => null); if (!body || typeof body !== "object") return json(400, { ok: false, error: "Invalid JSON body" }); // ---- Search broadly (Phase 3) ---- if ((body as any).action === "search_by_name") { const { event_id, full_name } = body as any; const { data, error } = await admin.rpc("door_search_candidates", { p_event_id: event_id, p_query: full_name, }); if (error) return json(500, { ok: false, error: error.message }); // door_search_candidates returns jsonb return (data as any)?.ok ? json(200, data as any) : json(409, data as any); } // ---- Confirm explicitly (Phase 3) ---- if ((body as any).action === "confirm_candidate") { const { registration_id, event_id, phone_last4, email_prefix, method } = body as any; const { data: v, error: ve } = await admin.rpc("door_verify_identity", { p_registration_id: registration_id, p_event_id: event_id, p_phone_last4: phone_last4, p_email_prefix: email_prefix, }); if (ve) return json(500, { ok: false, error: ve.message }); // door_verify_identity historically returned boolean; newer versions return JSON: // { ok: true/false, verifier: "email_prefix" | "phone_last4", code?: string, ... } // Normalize here so the rest of the flow is stable. let verifyPayload: any = v; // If it came back as a JSON string, try parse. if (typeof verifyPayload === "string") { try { verifyPayload = JSON.parse(verifyPayload); } catch { // leave as string } } if (typeof verifyPayload === "boolean") { verifyPayload = { ok: verifyPayload }; } const verifyOk = !!verifyPayload && typeof verifyPayload === "object" && !!verifyPayload.ok; if (!verifyOk) { // Pass through verification payload/codes (door.html uses phone_missing_use_email_prefix) // Fallback if payload isn't an object. return json(409, (typeof verifyPayload === "object" && verifyPayload) ? verifyPayload : { ok: false, code: "not_verified" }); } const { data: checkin, error: ce } = await admin.rpc("door_checkin_atomic", { p_registration_id: registration_id, p_method: method ?? "manual", }); if (ce) return json(500, { ok: false, error: ce.message }); return (checkin as any)?.ok ? json(200, checkin as any) : json(409, checkin as any); } // ---- QR path (existing) ---- if ((body as any).action === "qr" || ((body as any).registration_id && (body as any).method && !(body as any).action)) { const { registration_id, method } = body as any; const { data, error } = await admin.rpc("door_checkin_atomic", { p_registration_id: registration_id, p_method: method, }); if (error) return json(500, { ok: false, error: error.message }); return (data as any)?.ok ? json(200, data as any) : json(409, data as any); } // ---- Manual exact path (existing fallback) ---- if ((body as any).action === "manual" || ((body as any).email && (body as any).full_name)) { const { email: guestEmail, full_name, event_id } = body as any; const { data: lookup, error: le } = await admin.rpc("door_lookup_registration", { p_email: guestEmail, p_full_name: full_name, p_event_id: event_id ?? null, }); if (le) return json(500, { ok: false, error: le.message }); if (!(lookup as any)?.ok) return json(409, lookup as any); const { data: checkin, error: ce } = await admin.rpc("door_checkin_atomic", { p_registration_id: (lookup as any).registration_id, p_method: "manual", }); if (ce) return json(500, { ok: false, error: ce.message }); return (checkin as any)?.ok ? json(200, { ...(checkin as any), flow: "manual_lookup_then_checkin", name_match: (lookup as any).name_match, name_on_file: (lookup as any).name_on_file, }) : json(409, checkin as any); } return json(400, { ok: false, error: "Invalid request" }); } catch { return json(500, { ok: false, error: "Unexpected server error" }); } });
